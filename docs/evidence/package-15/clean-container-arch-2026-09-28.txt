#!/bin/bash
set -u
cd /work
echo "== container: $(. /etc/os-release; echo $PRETTY_NAME), $(ldd --version | head -1)"
pacman -Sy --noconfirm --needed zstd >/dev/null 2>&1
sha256sum -c pigment-prose-0.1.0-x86_64-linux.tar.zst.sha256
tar --zstd -xf pigment-prose-0.1.0-x86_64-linux.tar.zst -C /tmp
/tmp/pigment-prose-0.1.0-x86_64-linux/install.sh --prefix /opt/pp | tail -3
/opt/pp/bin/pigment-prose --version; /opt/pp/bin/pigment-studio --version
echo "== gpu-info with no Vulkan loader installed"
/opt/pp/bin/pigment-prose gpu-info; echo "exit $?"
echo "== installing only vulkan-icd-loader and vulkan-intel"
pacman -S --noconfirm vulkan-icd-loader vulkan-intel >/dev/null 2>&1; echo "pacman exit $?"
pacman -Q vulkan-icd-loader vulkan-intel mesa
/opt/pp/bin/pigment-prose gpu-info; echo "exit $?"
/opt/pp/bin/pigment-prose export --recipe 01.recipe.json --size 8k --adapter intel --out /work/ctr-8k.png; echo "export exit $?"
/opt/pp/share/pigment-prose/uninstall.sh
ls /opt/pp/bin 2>/dev/null | wc -l
